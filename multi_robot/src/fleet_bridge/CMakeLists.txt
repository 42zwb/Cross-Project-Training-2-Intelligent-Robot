cmake_minimum_required(VERSION 3.16)
project(fleet_bridge)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros_gz_bridge REQUIRED)

add_executable(parameter_bridge src/parameter_bridge.cpp)
target_include_directories(parameter_bridge PRIVATE include)
ament_target_dependencies(parameter_bridge rclcpp ros_gz_bridge)

add_executable(bridge_lifecycle_probe src/bridge_lifecycle_probe.cpp)
target_include_directories(bridge_lifecycle_probe PRIVATE include)
ament_target_dependencies(bridge_lifecycle_probe rclcpp ros_gz_bridge)
add_executable(bridge_traffic_source src/bridge_traffic_source.cpp)
ament_target_dependencies(bridge_traffic_source ros_gz_bridge)
add_executable(signal_join_probe src/signal_join_probe.cpp)
ament_target_dependencies(signal_join_probe rclcpp)
install(TARGETS parameter_bridge bridge_lifecycle_probe bridge_traffic_source signal_join_probe DESTINATION lib/${PROJECT_NAME})
install(FILES UPSTREAM.md DESTINATION share/${PROJECT_NAME})
ament_package()
